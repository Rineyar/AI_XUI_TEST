#Сборка Rust
#==========================================#
#FROM rust:1.99.0-trixie AS rust-builder TODO обновиться позже
FROM rust:1.98.1-trixie AS rust-builder

#Py среда для сборки и линковки
RUN apt-get update \
    && apt-get install -y --no-install-recommends python3.13-dev \
    && rm -rf /var/lib/apt/lists/*  

WORKDIR /build/agent/agent_core

#Исходники для ядра
COPY agent/agent_core /build/agent/agent_core

#Подсосы
COPY agent/skills /build/agent/skills
COPY agent/tools /build/agent/tools
COPY agent/system_prompt.md /build/agent/system_prompt.md
COPY agent/guardrails /build/agent/guardrails

#Собрать без сохранения
##Обратная совместимость присутствует. Желательно закоментить это и раскоментить нормльную сборку
# RUN cargo build --release && cp target/release/agent_core /build/agent_core

#Собрать со схранением
RUN --mount=type=cache,id=cargo-registry,target=/usr/local/cargo/registry \
    --mount=type=cache,id=cargo-git,target=/usr/local/cargo/git \
    --mount=type=cache,id=agent-target,target=/build/agent/agent_core/target \
    cargo build --release \
    #--features unsafe_tools \
    && cp /build/agent/agent_core/target/release/agent_core /build/agent_core

#==========================================#

#Сборка Nuclei
#==========================================#
FROM golang:1.26-alpine AS nuclei-builder

RUN apk add --no-cache git

RUN CGO_ENABLED=0 go install -v github.com/projectdiscovery/nuclei/v3/cmd/nuclei@v3.11.1
#==========================================#

#Загрузка Nuclei Templates
#==========================================#
FROM alpine:3.22 AS nuclei-templates-builder

RUN apk add --no-cache git ca-certificates

RUN git clone --depth 1 --branch v10.5.0 https://github.com/projectdiscovery/nuclei-templates.git /nuclei-templates \
    && rm -rf /nuclei-templates/.git
#==========================================#

#Запуск агента
#==========================================#
FROM python:3.13-slim-trixie

#Имя
LABEL description="Agent runtime test container"

#Переменные среды
ENV PYTHONUNBUFFERED=1
ENV RUST_BACKTRACE=1
ENV LC_ALL=C.UTF-8
ENV NUCLEI_TEMPLATES_DIR=/agent/workspace/nuclei-templates
ENV ZAP_PLANS_DIR=/agent/workspace/zap_plans
ENV AUTO=0
ENV UNSAFE=1

#Создание пользователя и Изолятора
RUN useradd --create-home --uid 1000 --user-group agent && install -d -o root -g agent -m 1770 /agent/workspace

#Добавить необходимые библиотеки питону
RUN pip install --no-cache-dir \
    requests==2.34.2 \
    bandit==1.9.4 \
    semgrep==1.180.0 \
    && pip check

#Установка пакетов для загрузки по http
RUN apt-get update && apt-get install -y --no-install-recommends \
    ca-certificates \
    && rm -rf /var/lib/apt/lists/*

#Nuclei с доступом из path
COPY --from=nuclei-builder /go/bin/nuclei /usr/local/bin/nuclei

#Взять собранный под контейнер ELF
COPY --from=rust-builder /build/agent_core /agent/agent_core/agent_core

#Рабочая область
WORKDIR /agent

#Его файлы на тест

# COPY --chown=root:root tests /agent/workspace/tests

# Для Zap
COPY --from=docker:cli /usr/local/bin/docker /usr/local/bin/docker
COPY --chown=root:root zap-plans /agent/workspace/zap-plans

# Шаблоны nuclei
COPY --from=nuclei-templates-builder nuclei-templates /agent/workspace/nuclei-templates

#Стартуем
#==========================================#

#Скрипт запуска + авточистка
COPY entrypoint.sh /usr/local/bin/entrypoint.sh
COPY cleanup.sh /usr/local/bin/cleanup.sh
